import json
import random
from datasets import load_dataset

# ===== הגדרות =====
HF_DATASET = "BeIR/fiqa"
HF_CONFIG  = "corpus"
SPLIT      = "corpus"
TEXT_COL   = "text"
TITLE_COL  = "title"  # שונה ל-title מכיוון שב-fiqa יש כותרות
MAX_DOCS   = 3000
# ===================

def load_my_corpus():
    ds = load_dataset(HF_DATASET, HF_CONFIG, split=SPLIT)
    print("עמודות:", ds.column_names, "| שורות:", f"{len(ds):,}")

    if TEXT_COL not in ds.column_names:
        raise KeyError(f"אין עמודה '{TEXT_COL}'. בחרו אחת מ: {ds.column_names}")

    ds = ds.shuffle(seed=42).select(range(min(MAX_DOCS, len(ds))))
    docs = []

    for i, row in enumerate(ds):
        text = row[TEXT_COL]
        if not isinstance(text, str) or not text.strip():
            continue
        title = str(row[TITLE_COL]) if TITLE_COL and row[TITLE_COL] else ""
        # שמירת המזהה המקורי או מזהה רץ
        doc_id = str(row.get("_id", f"doc{i:05d}"))
        docs.append({"id": doc_id, "title": title, "text": text})

    return docs

my_docs = load_my_corpus()
